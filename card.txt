import { Component, Input } from '@angular/core';
import { CommonModule } from '@angular/common';
import { MatIconModule } from '@angular/material/icon';
import { MatRippleModule } from '@angular/material/core';

export interface PriceRequest {
  requestID: number;
  requestedDate: string;
  requestStatus: { statusCode: string; subStatusCode: string };
  offerExpirationDate: string | null;
  origin: { cityName: string; stateAbbreviation: string };
  destination: { cityName: string; stateAbbreviation: string };
  commodityDescription: string;
  stcc: string;
}

@Component({
  selector: 'app-price-requests',
  standalone: true,
  imports: [CommonModule, MatIconModule, MatRippleModule],
  templateUrl: './price-requests.component.html',
  styleUrls: ['./price-requests.component.scss']
})
export class PriceRequestsComponent {
  @Input() requests: PriceRequest[] = [];

  getStatusBadges(req: PriceRequest): { label: string; cssClass: string }[] {
    const badges = [];
    const code = req.requestStatus?.statusCode;
    if (code === 'PA') badges.push({ label: 'Pending Acceptance', cssClass: 'badge-pending' });
    if (code === 'POR') badges.push({ label: 'Pending Other Railroad', cssClass: 'badge-other-rr' });
    if (code === 'NEW') badges.push({ label: 'New', cssClass: 'badge-new' });
    if (code === 'IW') badges.push({ label: 'In Work', cssClass: 'badge-inwork' });
    if (req.offerExpirationDate) {
      const days = this.daysUntil(req.offerExpirationDate);
      badges.push({ label: `Expires in ${days} days`, cssClass: 'badge-expires' });
    }
    return badges;
  }

  daysUntil(dateStr: string): number {
    const diff = new Date(dateStr).getTime() - Date.now();
    return Math.ceil(diff / (1000 * 60 * 60 * 24));
  }
}
